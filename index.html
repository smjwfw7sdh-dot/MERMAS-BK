function actualizarResumen() { const hoy = new Date().toISOString().split('T')[0]; const delDia = mermasData.filter(i => i.fecha === hoy); let totalMonto = 0; delDia.forEach(i => { totalMonto += (parseFloat(i.monto) || 0); }); const totalElement = document.getElementById("summaryTotal"); const formateado = totalMonto.toLocaleString('es-CL', { style: 'currency', currency: 'CLP', maximumFractionDigits: 0 }); totalElement.textContent = formateado; totalElement.classList.remove('rojo', 'verde'); if (totalMonto >= 50000) { totalElement.classList.add('rojo'); } else { totalElement.classList.add('verde'); } // Guardamos el total del día para usarlo en la Lupa window.totalMermaHoy = totalMonto; calcularLupa(); } function calcularLupa() { const totalMerma = window.totalMermaHoy || 0; const valorBase = parseFloat(document.getElementById("valorBase").value) || 0; const difTotal = parseFloat(document.getElementById("difTotalLupa").value); const porcentajeEl = document.getElementById("porcentajeMerma"); const estadoEl = document.getElementById("estadoLupa"); if (valorBase <= 0) { porcentajeEl.textContent = "0,00%"; estadoEl.textContent = "Ingresa el valor base"; estadoEl.className = "estado-lupa"; return; } // Cálculo del % de merma const porcentaje = (totalMerma / valorBase) * 100; porcentajeEl.textContent = porcentaje.toFixed(2).replace('.', ',') + "%"; // Estado según el % de merma if (porcentaje < 1) { estadoEl.textContent = "✅ Merma controlada (< 1%)"; estadoEl.className = "estado-lupa bueno"; } else if (porcentaje < 2.5) { estadoEl.textContent = "⚠️ Merma en rango aceptable"; estadoEl.className = "estado-lupa alerta"; } else { estadoEl.textContent = "🔴 Merma alta - Revisar"; estadoEl.className = "estado-lupa critico"; } // Si hay Dif Total ingresado, damos un comentario extra if (!isNaN(difTotal)) { if (Math.abs(porcentaje - Math.abs(difTotal)) < 0.5) { estadoEl.textContent += " | Las mermas explican la diferencia"; } else if (porcentaje < Math.abs(difTotal)) { estadoEl.textContent += " | Hay diferencia no justificada por mermas"; } } }
